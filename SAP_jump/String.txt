def permutation(s, l, n, result):
    if l == n:
        word = "".join(s)
        if ispalindrome(word) and word not in result:
            result.append(word)
    else:
        for i in range(l, n + 1):
            # swap
            s[l], s[i] = s[i], s[l]
            permutation(s, l + 1, n, result)
            # backtrack (undo swap)
            s[l], s[i] = s[i], s[l]
            
def ispalindrome(word):
    # manually check palindrome
    i = 0
    j = len(word) - 1
    while i < j:
        if word[i] != word[j]:
            return False
        i += 1
        j -= 1
    return True
